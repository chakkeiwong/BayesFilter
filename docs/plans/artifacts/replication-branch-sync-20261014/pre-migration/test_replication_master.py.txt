def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


runner = load("replication_master", ROOT / "scripts/run_replication_master.py")
worker = load("replication_diagnostic", ROOT / "docs/benchmarks/diagnose_replication_stage0.py")

class MasterTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory()
        self.addCleanup(self.temp.cleanup)
        self.base = Path(self.temp.name)

    def campaign(self):
        return runner.Campaign(self.base / "campaign")

    def complete_recovery_fixture(self):
        campaign = self.campaign()
        output = campaign.output / 'trial'
        output.mkdir()
        phases = {label: {'status': 'PASS', 'epochs_completed': 1, 'epoch_cap': cap,
                          'termination': 'SOURCE_EARLY_STOP', 'source_skipped_gradients': 0}
                  for label, cap in [('pretraining', 400), ('full_training', 800)]}
        epochs = [{'phase': label, 'function': function, 'epoch': 0, 'loss': 1.0}
                  for label in phases for function in ['do_training_epoch', 'do_validation_epoch']]
        checkpoints = {}
        for label in phases:
            path = output / label / 'models' / 'fixture_best.pt'
            path.parent.mkdir(parents=True)
            path.write_bytes(b'fixture checkpoint: byte integrity only')
            checkpoints[str(path.relative_to(output))] = runner.digest(path)
        result = {'check': 'PASS', 'mode': 'source_trial', 'source_training_protocol_complete': True,
                  'full_source_training_call_chain': 'PASS_ON_COMPLETE_SOURCE_TRIAL',
                  'phases': phases, 'optimizer_gradient_checks': 2, 'checkpoints_reloaded': 2}
        runner.write_json(output / 'result.json', result)
        runner.write_json(output / 'progress.json', {'terminal': True, 'phases': phases})
        runner.write_json(output / 'completed-phases.json', {'phases': phases, 'epochs': epochs})
        runner.write_json(output / 'source-training-wiring.json',
                          {'checkpoint_sha256': checkpoints, 'optimizer_gradient_checks': 2})
        return campaign, output, result

    def recovery_args(self):
        return types.SimpleNamespace(after_attempt=1, lost_supervisor_pid=999999999,
                                     continuation_plan=self.base / 'plan.md', repair_note='CPU test')

    def recovery_target(self, campaign, output, pid=999999998):
        target = {'index': 1, 'name': 'mdpf_source_trial', 'resource': 'gpu',
                  'status': 'RUNNING', 'started_at': runner.now(), 'timeout_seconds': 60,
                  'pid': pid, 'command': ['fixture'], 'output': str(output)}
        campaign.state['attempts'].append(target)
        campaign.save()
        return target

    def test_recovery_accepts_complete_records_and_rejects_corrupted_checkpoint(self):
        campaign, output, result = self.complete_recovery_fixture()
        self.assertEqual(runner.recovered_source_artifacts(output), result)
        checkpoint = output / 'full_training/models/fixture_best.pt'
        checkpoint.write_bytes(b'corrupted')
        with self.assertRaisesRegex(RuntimeError, 'checkpoint'):
            runner.recovered_source_artifacts(output)

    def test_recovery_rejects_partial_epoch_evidence_even_with_pass_label(self):

            runner.recovered_source_artifacts(output)

    def test_recovery_rejects_short_trial_and_nonterminal_progress(self):

                campaign, output, result = self.complete_recovery_fixture()

                # Reuse the test root for the next independently named fixture.
                self.base = self.base / 'next'

    def test_recovery_never_fabricates_exit_code_or_restarts_training(self):
        campaign, output, result = self.complete_recovery_fixture()
        target = self.recovery_target(campaign, output)
        identity = {'state': 'S', 'group': target['pid'], 'start_ticks': 1, 'command': ['fixture']}
        with mock.patch.object(runner, 'process_record', side_effect=[None, identity, None]), \
             mock.patch.object(runner.os, 'killpg'), \
             mock.patch.object(runner.subprocess, 'run', return_value=subprocess.CompletedProcess([], 0)) as followup:
            self.assertEqual(runner.supervise_existing(self.recovery_args(), campaign.output, self.base), 0)
        saved = json.loads(campaign.state_path.read_text())['attempts'][0]
        self.assertEqual(saved['status'], 'RECOVERED_ARTIFACTS')
        self.assertIsNone(saved['returncode'])
        self.assertEqual(saved['exit_status'], 'UNKNOWN_NONCHILD')
        followup.assert_called_once()
        self.assertIn('followup-evaluation', followup.call_args.args[0])
        self.assertNotIn('mdpf_source_trial', followup.call_args.args[0])

    def test_recovery_observed_failure_vetoes_pass_artifacts(self):
        campaign, output, result = self.complete_recovery_fixture()
        target = self.recovery_target(campaign, output)
        identity = {'state': 'S', 'group': target['pid'], 'start_ticks': 1, 'command': ['fixture']}
        zombie = {**identity, 'state': 'Z', 'exit_status': 42 << 8}
        with mock.patch.object(runner, 'process_record', side_effect=[None, identity, zombie]), \
             mock.patch.object(runner.os, 'killpg'), mock.patch.object(runner.subprocess, 'run') as followup:
            self.assertEqual(runner.supervise_existing(self.recovery_args(), campaign.output, self.base), 2)
        saved = json.loads(campaign.state_path.read_text())['attempts'][0]
        self.assertEqual((saved['status'], saved['returncode']), ('FAILED', 42))
        followup.assert_not_called()

    def test_recovery_enforces_original_deadline_and_charges_original_start(self):
        campaign, output, result = self.complete_recovery_fixture()
        command = [sys.executable, '-c', 'import time; time.sleep(60)']
        child = subprocess.Popen(command, start_new_session=True)
        try:
            target = self.recovery_target(campaign, output, child.pid)
            target.update(command=command, timeout_seconds=1,
                          started_at=runner.datetime.fromtimestamp(time.time()-60, runner.timezone.utc).isoformat())
            campaign.save()
            with mock.patch.object(runner.subprocess, 'run') as followup:
                self.assertEqual(runner.supervise_existing(self.recovery_args(), campaign.output, self.base), 2)
            child.wait(timeout=5)
            self.assertEqual(child.returncode, -9)
            saved = json.loads(campaign.state_path.read_text())['attempts'][0]
            self.assertEqual(saved['status'], 'TIMEOUT')
            self.assertGreaterEqual(saved['charged_seconds'], 60)
            self.assertLess(saved['charged_seconds'], 65)
            followup.assert_not_called()
        finally:
            if child.poll() is None:
                child.kill()
                child.wait(timeout=5)

    def test_timeout_stops_descendant_and_is_charged(self):
        campaign = self.campaign()
        pidfile = self.base / "child.pid"
        source = ("import subprocess,sys,time,pathlib; "
                  "p=subprocess.Popen([sys.executable,'-c','import time;time.sleep(60)']); "
                  "pathlib.Path(sys.argv[1]).write_text(str(p.pid)); time.sleep(60)")
        attempt = campaign.launch("timeout", "cpu",
                                  lambda out: [sys.executable, "-c", source, str(pidfile)],
                                  cwd=ROOT, timeout=1.0)
        self.assertEqual(attempt["status"], "TIMEOUT")
        self.assertGreater(attempt["charged_seconds"], .9)
        pid = int(pidfile.read_text())
        proc = Path("/proc") / str(pid) / "stat"
        # A killed child may remain a zombie until the system reaps it.
        for _ in range(100):
            if not proc.exists() or proc.read_text().split()[2] == "Z":
                break
            import time
            time.sleep(.01)
        self.assertTrue(not proc.exists() or proc.read_text().split()[2] == "Z")
        restored = runner.Campaign(campaign.output, resume=True)
        self.assertEqual(len(restored.state["attempts"]), 1)
        self.assertLess(restored.remaining("cpu"), runner.LIMITS["cpu_seconds"])

    def test_authorized_budget_extension_preserves_charges_and_limits(self):
        campaign = self.campaign()
        campaign.state["attempts"] = [{
            "name": "prior", "index": 1, "status": "COMPLETED", "resource": "gpu",
            "timeout_seconds": 300., "charged_seconds": 155.,
            "outcome": {"check": "PASS"}}]
        before = campaign.state["limits"].copy()

        restored = runner.Campaign(campaign.output, resume=True)
        self.assertEqual(restored.remaining("gpu"), after["gpu_seconds"] - 155.)
        self.assertEqual(restored.remaining("cpu"), before["cpu_seconds"])
        result = runner.report(restored)
        self.assertEqual(result["attempts_remaining"], 59)
        self.assertIn("Attempts: 1 / 60.", (campaign.output / "result.md").read_text())
        self.assertEqual(runner.LIMITS["gpu_seconds"], 900.)

    def test_reuse_requires_unchanged_dependencies(self):
        dependencies = {"source": "v1", "interpreter": "environment1"}
        fingerprint = runner.content_identity(dependencies)
        attempt = {"dependency_fingerprint": fingerprint, "outcome": {"check": "PASS"}}
        self.assertTrue(runner.reusable(attempt, fingerprint))
        for key in dependencies:
            changed = dict(dependencies)
            changed[key] += "-changed"
            self.assertFalse(runner.reusable(attempt, runner.content_identity(changed)))
        self.assertFalse(runner.reusable({"outcome": {"check": "PASS"}}, fingerprint))
        self.assertFalse(runner.reusable(attempt, None))
        attempt["outcome"]["check"] = "INCOMPLETE"
        self.assertTrue(runner.reusable(attempt, fingerprint))
        self.assertFalse(runner.reusable(attempt, fingerprint, retry_incomplete=True))

    def test_shared_environment_failure_stops_dependent_probes(self):
        output = self.base / "common_failure"
        def prepare(campaign, evidence_root, **kwargs):
            manifest = {"author_python": sys.executable,
                        "job_fingerprints": {"broken": "a", "dependent": "b"}}
            runner.write_json(campaign.output / "manifest.json", manifest)
            campaign.state["active_manifest"] = "manifest.json"
            campaign.save()
            return manifest
        failure = ("import json,pathlib,sys; "
                   "pathlib.Path(sys.argv[1]).write_text(json.dumps({"
                   "'check':'FAILED','classification':'reference_environment_failure'})); sys.exit(1)")
        builders = {
            "broken": lambda out: [sys.executable, "-c", failure, out / "result.json"],
            "dependent": lambda out: [sys.executable, "-c", "raise RuntimeError('must not launch')"]}
        argv = ["runner", "run", "--output", str(output), "--evidence-root", str(ROOT)]
        with mock.patch.object(sys, "argv", argv), mock.patch.object(runner, "prepare", prepare), \
             mock.patch.object(runner, "JOBS", (("broken", "cpu"), ("dependent", "cpu"))), \
             mock.patch.object(runner, "dependency_inputs", return_value={}), \
             mock.patch.object(runner, "job_fingerprints", return_value={"broken": "a", "dependent": "b"}), \
             mock.patch.object(runner, "build_jobs", return_value=builders):
            self.assertEqual(runner.main(), 2)
        ledger = json.loads((output / "ledger.json").read_text())
        self.assertEqual(len(ledger["attempts"]), 1)
        self.assertEqual(ledger["attempts"][0]["outcome"]["classification"], "reference_environment_failure")
